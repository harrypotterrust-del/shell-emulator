# Этап 3: проверка всех команд, реализованных на этапах 1-3.
# Скрипт запускается с каждым вариантом VFS (scripts/run_vfs_*.sh).

# Раскрытие переменных окружения: в сообщении об ошибке виден
# раскрытый путь, неизвестная переменная даёт пустую строку.
ls $HOME
ls '$HOME' "${USER} files"
ls $NOPE
cd "my folder"
cd my\ dir

# Команды работают с корнем VFS.
ls -a
cd /
ls

# Служебная команда просмотра загруженной VFS.
vfs-tree

# Обработка ошибок: все виды сразу.
vfs-tree лишний аргумент
foo bar
cd a b
ls 'незакрытая кавычка
ls ${1x}
exit abc
